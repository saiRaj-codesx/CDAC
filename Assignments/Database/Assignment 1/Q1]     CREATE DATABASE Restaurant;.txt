Q1]     CREATE DATABASE Restaurant;
	USE Restaurant;


	CREATE TABLE Category(
	category_id INT,
	category_name VARCHAR(50)
	);


	CREATE TABLE Dish(
	dish_id INT,
	dish_name VARCHAR(50),
	category_id INT,
	price DECIMAL(10,2),
	dish_rank VARCHAR(50)
	);

        CREATE TABLE Customer(
	customer_id INT,
	customer_name VARCHAR(50),
	phone VARCHAR(50)
	);

        CREATE TABLE Orders(
	order_id INT,
	customer_id INT,
	order_date DATE
	);

        CREATE TABLE OrderSpecs(
	specs_id INT,
        order_id INT,
        dish_id INT,
        quantity INT
	);
