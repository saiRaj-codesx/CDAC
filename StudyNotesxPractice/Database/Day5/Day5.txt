Multi-row subquery: 

select emp_name, salary, dpet_id from employee
where dpet_id in  (select dpet_id from department where  dept_name in ('IT', 'Sales'));

multi-column subquery: 

return two or more columns; 

select emp_name, salary, dpet_id 
from employee 
where (salary, dpet_id) In   (select Min(salary),dpet_id 
from employee group by dpet_id);


where subquery : 


In, Any , ALL : 
IN : checks whether a value matches any value from returned result by the subquery:

Any : the comparison is true if it is true for at least one returned value.

select emp_name, salary
from employee 
where salary > Any select salary from employee where dpet_id=20);


ALL

select emp_name, salary
from employee 
where salary > All (select salary from employee where dpet_id=10);

All: comparison must be true for every returned value;


correlated subquery: a correlated subquery refers to a column from outer query;
select e.emp_name, e.salary, e.dpet_id
from employee e
where e.salary >  ( select avg(salary) from employee e2 
where e.dpet_id= e2.dept_id);
